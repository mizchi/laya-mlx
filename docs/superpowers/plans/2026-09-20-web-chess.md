Tests: vitest for candidates/policy/game (known FENs), Playwright with the stub agent (user move → AI reply) and an env-gated real-model game.

**Tech Stack:** chess.js 1.4, TypeScript, Vite, vitest, Playwright, `@laya-mlx/web`.

```ts
import { Chess, type Color, type Square } from "chess.js";

export type { Color, Square };
export interface MoveInput {
  from: string;
  to: string;
  promotion?: "q" | "r" | "b" | "n";
}
export interface GameStatus {
  over: boolean;
  text: string;
}

/** Thin, rule-complete wrapper over chess.js with the demo's notion of a user side. */
export class ChessGame {
  readonly chess: Chess;

  constructor(
    fen?: string,
    readonly userColor: Color = "w",
  ) {
    this.chess = fen ? new Chess(fen) : new Chess();
  }

  get turn(): Color {
    return this.chess.turn();
  }
  get isUserTurn(): boolean {
    return this.turn === this.userColor && !this.status.over;
  }
  get history(): string[] {
    return this.chess.history();
  }
  get fen(): string {
    return this.chess.fen();
  }

  /** chess.js board(): rank 8 first; each cell `{square, type, color}` or null. */
  board() {
    return this.chess.board();
  }

  legalTargets(from: string): string[] {
    return this.chess.moves({ square: from as Square, verbose: true }).map((m) => m.to);
  }

  /** Apply a SAN string or a from/to pair (promotion defaults to queen). Returns the SAN or null if illegal. */
  applyMove(move: string | MoveInput): string | null {
    try {
      const played =
        typeof move === "string"
          ? this.chess.move(move)
          : this.chess.move({ from: move.from, to: move.to, promotion: move.promotion ?? "q" });
      return played.san;
    } catch {
      return null;
    }
  }

  get status(): GameStatus {
    const c = this.chess;
    const side = c.turn() === "w" ? "White" : "Black";
    const other = c.turn() === "w" ? "Black" : "White";
    if (c.isCheckmate()) return { over: true, text: `Checkmate — ${other} wins` };
    if (c.isStalemate()) return { over: true, text: "Stalemate — draw" };
    if (c.isThreefoldRepetition()) return { over: true, text: "Draw by repetition" };
    if (c.isInsufficientMaterial()) return { over: true, text: "Draw — insufficient material" };
    if (c.isDraw()) return { over: true, text: "Draw" };
    return { over: false, text: c.inCheck() ? `${side} to move — check` : `${side} to move` };
  }
}
```

chess.js 1.x throws on illegal moves — hence the try/catch. Keep chess.js as the only rules source; no move generation elsewhere.

```ts
import type { Chess, Move } from "chess.js";

export const MATE = 1000;
export const SHORTLIST = 6;
const VALUE: Record<string, number> = { p: 1, n: 3, b: 3, r: 5, q: 9, k: 0 };
const NAME: Record<string, string> = { p: "pawn", n: "knight", b: "bishop", r: "rook", q: "queen", k: "king" };

/** Material balance in pawns from the side to move's perspective. */
export function evaluateMaterial(chess: Chess): number {
  const me = chess.turn();
  let total = 0;
  for (const row of chess.board()) {
    for (const cell of row) {
      if (cell) total += (cell.color === me ? 1 : -1) * VALUE[cell.type]!;
    }
  }
  return total;
}

/**
 * Score a move for the side to move: material after the opponent's best material reply
 * (2 plies), with checkmate/stalemate terminal values and small positional bonuses.
 */
export function scoreMove(chess: Chess, move: Move): { score: number; reply: Move | null; allowsMate: boolean } {
  chess.move(move.san);
  try {
    if (chess.isCheckmate()) return { score: MATE, reply: null, allowsMate: false };
    if (chess.isDraw()) return { score: 0, reply: null, allowsMate: false };
    let worst = Infinity;
    let worstReply: Move | null = null;
    let allowsMate = false;
    for (const reply of chess.moves({ verbose: true })) {
      chess.move(reply.san);
      let value: number;
      if (chess.isCheckmate()) {
        value = -MATE;
        allowsMate = true;
      } else if (chess.isDraw()) {
        value = 0;
      } else {
        // After the reply it is our turn again, so evaluateMaterial is from our perspective.
        value = evaluateMaterial(chess);
      }
      chess.undo();
      if (value < worst) {
        worst = value;
        worstReply = reply;
      }
    }
    return { score: worst + bonuses(chess, move), reply: worstReply, allowsMate };
  } finally {
    chess.undo();
  }
}

/** Small, explainable nudges; `chess` is positioned after `move`. */
function bonuses(chess: Chess, move: Move): number {
  let bonus = 0;
  if (chess.inCheck()) bonus += 0.1;
  if (move.flags.includes("k") || move.flags.includes("q")) bonus += 0.3;
  if ((move.piece === "n" || move.piece === "b") && isBackRank(move.from, move.color)) bonus += 0.2;
  if (move.piece === "p" && (move.to === "d4" || move.to === "e4" || move.to === "d5" || move.to === "e5")) bonus += 0.2;
  return bonus;
}

const isBackRank = (square: string, color: string) => square[1] === (color === "w" ? "1" : "8");

/** Snake-style template text; `chess` is positioned BEFORE the move. */
export function describeMove(chess: Chess, move: Move, materialDelta: number, allowsMate: boolean): string {
  const captured = move.captured ? NAME[move.captured]! : null;
  const piece = NAME[move.piece]!;
  chess.move(move.san);
  const mate = chess.isCheckmate();
  const check = chess.inCheck();
  chess.undo();
  if (mate) return "Checkmate. Best.";
  if (allowsMate) return "Allows checkmate. Unsafe.";
  const hangs = materialDelta <= -1 && !captured;
  const parts: string[] = [];
  if (captured) parts.push(`Captures the ${captured}.`);
  if (hangs) parts.push(`Hangs the ${piece}${hangsTo(move) ? ` to a ${hangsTo(move)}` : ""}.`);
  if (move.flags.includes("k")) parts.push("Castles kingside.");
  if (move.flags.includes("q")) parts.push("Castles queenside.");
  if (check) parts.push("Gives check.");
  if (!captured && !hangs && (move.piece === "n" || move.piece === "b") && isBackRank(move.from, move.color)) {
    parts.push(`Develops the ${piece}.`);
  }
  if (parts.length === 0) parts.push("Quiet move.");
  parts.push(materialDelta < 0 ? "Loses material." : "Safe.");
  return parts.join(" ");
}
```

```ts
export interface Candidate {
  san: string;
  from: string;
  to: string;
  score: number;
  description: string;
  best: boolean;
  allowsMate: boolean;
}

/** Top candidates by 2-ply material score with descriptions; exactly one is marked best. */
export function shortlist(chess: Chess, limit = SHORTLIST): Candidate[] {
  const before = evaluateMaterial(chess);
  const scored = chess.moves({ verbose: true }).map((move) => {
    const { score, reply, allowsMate } = scoreMove(chess, move);
    const materialDelta = Math.max(-MATE, Math.min(MATE, score)) - before;
    return { move, score, reply, allowsMate, materialDelta };
  });
  scored.sort((a, b) => b.score - a.score);
  const top = scored.slice(0, limit);
  return top.map((s, i) => ({
    san: s.move.san,
    from: s.move.from,
    to: s.move.to,
    score: s.score,
    description: describeMove(chess, s.move, s.materialDelta, s.allowsMate, s.reply) + (i === 0 ? " Best." : ""),
    best: i === 0,
    allowsMate: s.allowsMate,
  }));
}
```

Note "Checkmate. Best." already ends with "Best." — avoid doubling: only append " Best." when the description does not already end with "Best.". The bonus fraction makes `materialDelta` non-integer; use `Math.round` when comparing to `-1` in `describeMove` (`Math.round(materialDelta) <= -1`). Stable ordering: chess.js move order is deterministic, `sort` is stable in modern JS.

```ts
import type { PredictResult, Question, State } from "@laya-mlx/web";

import { type Candidate, MATE, shortlist } from "./candidates.ts";
import type { ChessGame } from "./game.ts";
import { evaluateMaterial } from "./candidates.ts";

export interface ChessAgent {
  predict(state: State, questions: Record<string, Question>): Promise<PredictResult>;
}

export interface Prompt {
  state: string;
  questions: Record<string, Question>;
}

export function buildPrompt(game: ChessGame, candidates: Candidate[]): Prompt {
  const material = evaluateMaterial(game.chess);
  const balance = material === 0 ? "even" : material > 0 ? `+${material}` : `${material}`;
  const moveNumber = game.chess.moveNumber();
  const color = game.turn === "w" ? "white" : "black";
  const check = game.chess.inCheck() ? "In check." : "Not in check.";
  return {
    state: `Chess. We play ${color}. Move ${moveNumber}. Material: ${balance}. ${check}`,
    questions: {
      move: {
        type: "choice",
        instructions: "Choose the strongest safe move.",
        criteria: Object.fromEntries(candidates.map((c) => [c.san, c.description])),
      },
      danger: { type: "noul", instructions: "Is our king in danger?" },
    },
  };
}

const SHIELD_MARGIN = 3;

export function applyShield(
  probabilities: Record<string, number>,
  candidates: Candidate[],
  guarded: boolean,
): { proposed: string; executed: string; intervened: boolean } {
  let proposed = candidates[0]!.san;
  for (const c of candidates) if ((probabilities[c.san] ?? -1) > (probabilities[proposed] ?? -1)) proposed = c.san;
  if (!guarded) return { proposed, executed: proposed, intervened: false };
  const best = candidates.find((c) => c.best) ?? candidates[0]!;
  const chosen = candidates.find((c) => c.san === proposed)!;
  const blunder = chosen.score < best.score - SHIELD_MARGIN;
  const mate = chosen.allowsMate && !best.allowsMate;
  const executed = blunder || mate ? best.san : proposed;
  return { proposed, executed, intervened: executed !== proposed };
}

export interface Decision {
  probabilities: Record<string, number>;
  proposed: string;
  executed: string;
  intervened: boolean;
  kingInDanger: number;
  inferenceMs: number;
  decisionMs: number;
  inputTokens: number;
  outputTokens: number;
  candidates: Candidate[];
}

export interface PolicyOptions {
  guarded: boolean;
}

export async function decide(agent: ChessAgent, game: ChessGame, options: PolicyOptions): Promise<Decision> {
  const started = performance.now();
  const candidates = shortlist(game.chess);
  if (candidates.length === 0) throw new Error("Chess: no legal moves");
  const prompt = buildPrompt(game, candidates);
  const inferenceStart = performance.now();
  const output = await agent.predict(prompt.state, prompt.questions);
  const inferenceMs = performance.now() - inferenceStart;
  const move = output.answers["move"];
  const danger = output.answers["danger"];
  if (move?.type !== "choice") throw new Error("Model returned an unexpected shape for the move answer");
  if (danger?.type !== "noul") throw new Error("Model returned an unexpected shape for the danger answer");
  const probabilities: Record<string, number> = {};
  for (const c of candidates) probabilities[c.san] = move.probabilities[c.san] ?? Number.NaN;
  const values = [...Object.values(probabilities), danger.noul];
  if (values.some((v) => !Number.isFinite(v) || v < 0 || v > 1)) {
    throw new Error("Model returned an invalid probability; no move executed");
  }
  const { proposed, executed, intervened } = applyShield(probabilities, candidates, options.guarded);
  return {
    probabilities,
    proposed,
    executed,
    intervened,
    kingInDanger: danger.noul,
    inferenceMs,
    decisionMs: performance.now() - started,
    inputTokens: output.usage.input_tokens,
    outputTokens: output.usage.output_tokens ?? 0,
    candidates,
  };
}
```

SAN strings like `Rxa8+` and `O-O` are fine as choice labels (they are strings; none are integer-like). `MATE` import is unused in the sketch — drop it or use it.
